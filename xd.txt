# ----------------- MODELOS -----------------


class User(SQLModel, table=True):
    id: Optional[int] = Field(default=None, primary_key=True)
    username: str = Field(index=True, unique=True)
    full_name: str
    hashed_password: str  # contraseña tal cual (por ahora)
    is_admin: bool = Field(default=False)


class Category(SQLModel, table=True):
    id: Optional[int] = Field(default=None, primary_key=True)
    name: str
    products: List["Product"] = Relationship(back_populates="category")


class Product(SQLModel, table=True):
    id: Optional[int] = Field(default=None, primary_key=True)
    name: str
    price: float
    category_id: int = Field(foreign_key="category.id")
    category: Optional[Category] = Relationship(back_populates="products")
    recipe_items: List["RecipeItem"] = Relationship(back_populates="product")


class Ingredient(SQLModel, table=True):
    id: Optional[int] = Field(default=None, primary_key=True)
    name: str
    unit: str = Field(default="pz")  # g, ml, pz, etc.
    stocks: List["IngredientInventory"] = Relationship(back_populates="ingredient")
    recipes: List["RecipeItem"] = Relationship(back_populates="ingredient")


class IngredientInventory(SQLModel, table=True):
    id: Optional[int] = Field(default=None, primary_key=True)
    ingredient_id: int = Field(foreign_key="ingredient.id")
    ingredient: Optional[Ingredient] = Relationship(back_populates="stocks")
    stock: float = Field(default=0)     # existencias
    min_stock: float = Field(default=0) # mínimo antes de marcar "bajo"


class RecipeItem(SQLModel, table=True):
    """Relación producto-ingrediente (receta): cuánto ingrediente usa 1 unidad del producto."""
    id: Optional[int] = Field(default=None, primary_key=True)
    product_id: int = Field(foreign_key="product.id")
    ingredient_id: int = Field(foreign_key="ingredient.id")
    quantity_per_unit: float  # ej: 100 g de arroz por roll

    product: Optional[Product] = Relationship(back_populates="recipe_items")
    ingredient: Optional[Ingredient] = Relationship(back_populates="recipes")


class Order(SQLModel, table=True):
    id: Optional[int] = Field(default=None, primary_key=True)
    order_type: str  # "COMEDOR", "LLEVAR", "DELIVERY"
    table_number: Optional[str] = None
    customer_name: Optional[str] = None
    delivery_platform: Optional[str] = None  # "RAPPI", "DIDI", "UBER", None
    items: List["OrderItem"] = Relationship(back_populates="order")


class OrderItem(SQLModel, table=True):
    id: Optional[int] = Field(default=None, primary_key=True)
    order_id: int = Field(foreign_key="order.id")
    product_id: int = Field(foreign_key="product.id")
    quantity: int
    unit_price: float
    order: Optional[Order] = Relationship(back_populates="items")
    product: Optional[Product] = Relationship()


class CashClose(SQLModel, table=True):
    """Cierre de caja: quién cerró, cuándo, cuánto había y diferencia."""
    id: Optional[int] = Field(default=None, primary_key=True)
    closed_at: datetime = Field(default_factory=datetime.utcnow)
    user_id: int = Field(foreign_key="user.id")
    total_sales: float
    reported_cash: float
    difference: float  # reported_cash - total_sales


# ----------------- SESIONES / LOGIN SIMPLE -----------------

sessions: Dict[str, int] = {}  # token -> user_id


def get_session():
    with Session(engine) as session:
        yield session


def get_current_user(request: Request, db: Session = Depends(get_session)) -> Optional[User]:
    token = request.cookies.get("session_id")
    if not token:
        return None
    user_id = sessions.get(token)
    if not user_id:
        return None
    user = db.get(User, user_id)
    return user


# ----------------- INICIALIZACIÓN BD + DATOS DEMO -----------------


def create_db_and_seed():
    SQLModel.metadata.create_all(engine)
    with Session(engine) as session:
        # usuario admin por defecto
        admin = session.exec(select(User).where(User.username == "admin")).first()
        if not admin:
            admin = User(
                username="admin",
                full_name="Administrador",
                hashed_password="admin",  # contraseña en texto plano
                is_admin=True,
            )
            session.add(admin)

        # categorías básicas
        if not session.exec(select(Category)).first():
            categorias = [
                "Entradas",
                "Platillos",
                "Sushis",
                "Bebidas",
                "Postres",
            ]
            for c in categorias:
                cat = Category(name=c)
                session.add(cat)
            session.commit()

            # productos demo
            productos_demo = [
                ("Edamames", 60, "Entradas"),
                ("Yakimeshi", 90, "Platillos"),
                ("Sushi California", 110, "Sushis"),
                ("Refresco Lata", 25, "Bebidas"),
                ("Helado Frito", 65, "Postres"),
            ]
            for nombre, precio, cat_nombre in productos_demo:
                cat = session.exec(select(Category).where(Category.name == cat_nombre)).first()
                if cat:
                    p = Product(name=nombre, price=precio, category_id=cat.id)
                    session.add(p)
            session.commit()

        # ingredientes demo
        if not session.exec(select(Ingredient)).first():
            ingredientes_demo = [
                ("Arroz para sushi", "g", 5000, 1000),
                ("Alga nori", "pz", 100, 20),
                ("Pepino", "g", 2000, 500),
                ("Surimi", "g", 2000, 500),
                ("Aceite para freír", "ml", 5000, 1000),
                ("Base helado", "g", 1000, 200),
                ("Refresco lata surtido", "pz", 48, 12),
                ("Edamame congelado", "g", 3000, 500),
            ]
            for nombre, unidad, stock, min_stock in ingredientes_demo:
                ing = Ingredient(name=nombre, unit=unidad)
                session.add(ing)
                session.flush()
                inv = IngredientInventory(
                    ingredient_id=ing.id,
                    stock=stock,
                    min_stock=min_stock,
                )
                session.add(inv)
            session.commit()

        # recetas demo por producto
        if not session.exec(select(RecipeItem)).first():
            recetas = {
                "Sushi California": [
                    ("Arroz para sushi", 120.0),
                    ("Alga nori", 1.0),
                    ("Pepino", 30.0),
                    ("Surimi", 40.0),
                ],
                "Yakimeshi": [
                    ("Arroz para sushi", 150.0),
                    ("Aceite para freír", 10.0),
                    ("Surimi", 30.0),
                ],
                "Edamames": [
                    ("Edamame congelado", 100.0),
                ],
                "Refresco Lata": [
                    ("Refresco lata surtido", 1.0),
                ],
                "Helado Frito": [
                    ("Base helado", 80.0),
                    ("Aceite para freír", 10.0),
                ],
            }

            for prod_name, lista_ings in recetas.items():
                producto = session.exec(
                    select(Product).where(Product.name == prod_name)
                ).first()
                if not producto:
                    continue
                for ing_name, qty in lista_ings:
                    ing = session.exec(
                        select(Ingredient).where(Ingredient.name == ing_name)
                    ).first()
                    if not ing:
                        continue
                    r = RecipeItem(
                        product_id=producto.id,
                        ingredient_id=ing.id,
                        quantity_per_unit=qty,
                    )
                    session.add(r)
            session.commit()


create_db_and_seed()

def render_template_string(template_str: str, **context):
    return HTMLResponse(content=Template(template_str).render(**context))


# ----------------- LOGIN / MENÚ -----------------


@app.get("/", response_class=HTMLResponse)
def root(request: Request, db: Session = Depends(get_session)):
    user = get_current_user(request, db)
    if user:
        return RedirectResponse(url="/menu", status_code=302)
    return render_template_string(LOGIN_HTML, error=None)


@app.get("/logout")
def logout():
    response = RedirectResponse(url="/", status_code=302)
    response.delete_cookie("session_id")
    return response


@app.post("/login", response_class=HTMLResponse)
def login(
    request: Request,
    username: str = Form(...),
    password: str = Form(...),
    db: Session = Depends(get_session),
):
    user = db.exec(select(User).where(User.username == username)).first()
    if not user or user.hashed_password != password:
        return render_template_string(LOGIN_HTML, error="Usuario o contraseña incorrectos")
    token = str(uuid.uuid4())
    sessions[token] = user.id
    response = RedirectResponse(url="/menu", status_code=302)
    response.set_cookie("session_id", token, httponly=True)
    return response


@app.get("/menu", response_class=HTMLResponse)
def menu(
    request: Request,
    db: Session = Depends(get_session),
):
    user = get_current_user(request, db)
    if not user:
        return RedirectResponse(url="/", status_code=302)
    return render_template_string(DASHBOARD_HTML, user=user)


# ----------------- POS / PEDIDOS -----------------


@app.get("/pos", response_class=HTMLResponse)
def pos(
    request: Request,
    db: Session = Depends(get_session),
):
    user = get_current_user(request, db)
    if not user:
        return RedirectResponse(url="/", status_code=302)

    categories = db.exec(select(Category).order_by(Category.name)).all()
    for c in categories:
        c.products

    low_inventory = db.exec(
        select(IngredientInventory).where(IngredientInventory.stock <= IngredientInventory.min_stock)
    ).all()

    return render_template_string(
        POS_HTML,
        user=user,
        categories=categories,
        last_order=None,
        last_total=0.0,
        ticket=None,
        kitchen=None,
        low_inventory=low_inventory,
        alert_message=None,
    )


def build_ticket(order: Order, items: List[OrderItem], total: float, item_options: Dict[int, str]) -> str:
    lines = []
    lines.append("------ SUSHI POS ------")
    lines.append(f"Pedido #{order.id}")
    lines.append(f"Tipo: {order.order_type}")
    if order.order_type == "COMEDOR":
        lines.append(f"Mesa: {order.table_number}")
    else:
        if order.customer_name:
            lines.append(f"Cliente: {order.customer_name}")
    if order.delivery_platform:
        lines.append(f"Plataforma: {order.delivery_platform}")
    lines.append("-----------------------")
    for it in items:
        lines.append(f"{it.quantity} x {it.product.name}  ${it.unit_price:.2f}")
        extra = item_options.get(it.product_id)
        if extra:
            lines.append(f"   -> {extra}")
    lines.append("-----------------------")
    lines.append(f"TOTAL: ${total:.2f}")
    lines.append("Gracias por su compra")
    return "\n".join(lines)


def build_kitchen_ticket(order: Order, items: List[OrderItem], item_options: Dict[int, str]) -> str:
    lines = []
    lines.append("**** COMANDA COCINA ****")
    lines.append(f"Pedido #{order.id} - {order.order_type}")
    if order.order_type == "COMEDOR":
        lines.append(f"Mesa: {order.table_number}")
    else:
        if order.customer_name:
            lines.append(f"Cliente: {order.customer_name}")
    if order.delivery_platform:
        lines.append(f"Plataforma: {order.delivery_platform}")
    lines.append("------------------------")
    for it in items:
        lines.append(f"{it.quantity} x {it.product.name}")
        extra = item_options.get(it.product_id)
        if extra:
            lines.append(f"   -> {extra}")
    lines.append("------------------------")
    lines.append("FIN COMANDA")
    return "\n".join(lines)


def write_ticket_files(order: Order, ticket: str, kitchen: str):
    os.makedirs("tickets_cliente", exist_ok=True)
    os.makedirs("comandas_cocina", exist_ok=True)
    with open(f"tickets_cliente/ticket_{order.id}.txt", "w", encoding="utf-8") as f:
        f.write(ticket)
    with open(f"comandas_cocina/comanda_{order.id}.txt", "w", encoding="utf-8") as f:
        f.write(kitchen)


@app.post("/orders/create", response_class=HTMLResponse)
async def create_order(
    request: Request,
    order_type: str = Form(...),
    reference: str = Form(""),
    delivery_platform: str = Form(""),
    db: Session = Depends(get_session),
):
    user = get_current_user(request, db)
    if not user:
        return RedirectResponse(url="/", status_code=302)

    form = await request.form()

    # 1) Leer lo que el usuario pidió (pero SIN guardar todavía en la BD)
    requested = []  # lista de dicts: {product, qty, options_text, skip_alga, skip_pepino}
    alert_message = None

    for key, value in form.items():
        if not key.startswith("qty_"):
            continue
        try:
            qty = int(value)
        except (ValueError, TypeError):
            continue
        if qty <= 0:
            continue

        product_id = int(key.split("_")[1])
        product = db.get(Product, product_id)
        if not product:
            continue

        # Opciones (alga, empanizado, notas)
        alga_opt = form.get(f"alga_{product_id}", "").strip()
        emp_opt = form.get(f"emp_{product_id}", "").strip()
        nota_opt_raw = form.get(f"nota_{product_id}", "").strip()
        nota_opt_lower = nota_opt_raw.lower()

        parts = []
        if alga_opt:
            parts.append(alga_opt)
        if emp_opt:
            parts.append(emp_opt)
        if nota_opt_raw:
            parts.append(nota_opt_raw)
        options_text = " / ".join(parts) if parts else ""

        # Flags para receta
        skip_alga = (alga_opt == "Sin alga")
        # si en las notas escribes "sin pepino" (con mayús/minus indistinto)
        skip_pepino = ("sin pepino" in nota_opt_lower)

        requested.append(
            {
                "product": product,
                "qty": qty,
                "options_text": options_text,
                "skip_alga": skip_alga,
                "skip_pepino": skip_pepino,
            }
        )

    # Preparar datos para recargar pantalla si algo falla
    categories = db.exec(select(Category).order_by(Category.name)).all()
    for c in categories:
        c.products
    low_inventory = db.exec(
        select(IngredientInventory).where(IngredientInventory.stock <= IngredientInventory.min_stock)
    ).all()

    if not requested:
        return render_template_string(
            POS_HTML,
            user=user,
            categories=categories,
            last_order=None,
            last_total=0.0,
            ticket=None,
            kitchen=None,
            low_inventory=low_inventory,
            alert_message="No agregaste ningún producto al pedido.",
        )

    # 2) Verificar si hay suficientes ingredientes para TODO lo que se pidió
    #    tomando en cuenta "Sin alga" y "sin pepino".
    shortages: Dict[str, bool] = {}  # nombre_ingrediente -> True (falta)

    for req in requested:
        prod = req["product"]
        qty = req["qty"]
        skip_alga = req["skip_alga"]
        skip_pepino = req["skip_pepino"]

        receta = db.exec(
            select(RecipeItem).where(RecipeItem.product_id == prod.id)
        ).all()

                for r in receta:
            ing = db.get(Ingredient, r.ingredient_id)
            if not ing:
                continue

            # reglas para omitir ingredientes según las opciones del cliente
            if skip_alga and "alga" in ing.name.lower():
                continue
            if skip_pepino and "pepino" in ing.name.lower():
                continue

            consumo = qty * r.quantity_per_unit
            inv = db.exec(
                select(IngredientInventory).where(
                    IngredientInventory.ingredient_id == ing.id
                )
            ).first()
            available = inv.stock if inv else 0
            if consumo > available:
                shortages[ing.name] = True

    if shortages:
        # Si falta al menos un ingrediente, NO creamos el pedido ni descontamos nada
        if len(shortages) == 1:
            nombre = list(shortages.keys())[0]
            alert_message = f"⚠ Cuidado: ya no tenemos suficiente {nombre} para este pedido."
        else:
            nombres = ", ".join(shortages.keys())
            alert_message = f"⚠ Cuidado: faltan ingredientes para este pedido: {nombres}."

        return render_template_string(
            POS_HTML,
            user=user,
            categories=categories,
            last_order=None,
            last_total=0.0,
            ticket=None,
            kitchen=None,
            low_inventory=low_inventory,
            alert_message=alert_message,
        )

    # 3) Si sí alcanza todo, ahora sí creamos la Orden y los OrderItem, y descontamos inventario
    order = Order(
        order_type=order_type,
        table_number=reference if order_type == "COMEDOR" else None,
        customer_name=reference if order_type != "COMEDOR" else None,
        delivery_platform=delivery_platform or None,
    )
    db.add(order)
    db.commit()
    db.refresh(order)

    items: List[OrderItem] = []
    total = 0.0
    item_options: Dict[int, str] = {}

    for req in requested:
        prod = req["product"]
        qty = req["qty"]
        options_text = req["options_text"]
        skip_alga = req["skip_alga"]
        skip_pepino = req["skip_pepino"]

        if options_text:
            item_options[prod.id] = options_text

        it = OrderItem(
            order_id=order.id,
            product_id=prod.id,
            quantity=qty,
            unit_price=prod.price,
        )
        db.add(it)
        items.append(it)
        total += qty * prod.price

        # Descontar ingredientes tomando en cuenta las opciones
        receta = db.exec(
            select(RecipeItem).where(RecipeItem.product_id == prod.id)
        ).all()
        for r in receta:
            ing = db.get(Ingredient, r.ingredient_id)
            if not ing:
                continue

            if skip_alga and "alga" in ing.name.lower():
                continue
            if skip_pepino and "pepino" in ing.name.lower():
                continue

            consumo = qty * r.quantity_per_unit
            inv = db.exec(
                select(IngredientInventory).where(
                    IngredientInventory.ingredient_id == ing.id
                )
            ).first()
            if inv:
                inv.stock = max(inv.stock - consumo, 0)


    db.commit()
    for it in items:
        db.refresh(it)
        db.refresh(it.product)

    ticket = build_ticket(order, items, total, item_options)
    kitchen = build_kitchen_ticket(order, items, item_options)
    write_ticket_files(order, ticket, kitchen)

    low_inventory = db.exec(
        select(IngredientInventory).where(IngredientInventory.stock <= IngredientInventory.min_stock)
    ).all()

    return render_template_string(
        POS_HTML,
        user=user,
        categories=categories,
        last_order=order,
        last_total=total,
        ticket=ticket,
        kitchen=kitchen,
        low_inventory=low_inventory,
        alert_message=None,
    )

# ----------------- UTILIDAD: TOTAL VENTAS -----------------


def calcular_total_ventas(db: Session) -> float:
    orders = db.exec(select(Order)).all()
    total = 0.0
    for o in orders:
        items = db.exec(select(OrderItem).where(OrderItem.order_id == o.id)).all()
        for it in items:
            total += it.quantity * it.unit_price
    return total

    user = get_current_user(request, db)

@app.post("/caja/cierre", response_class=HTMLResponse)
def caja_cierre_post(
    request: Request,
    reported_cash: float = Form(...),
    db: Session = Depends(get_session),
):
    user = get_current_user(request, db)
    if not user:
        return RedirectResponse(url="/", status_code=302)

    total_ventas = calcular_total_ventas(db)
    difference = reported_cash - total_ventas

    close = CashClose(
        user_id=user.id,
        total_sales=total_ventas,
        reported_cash=reported_cash,
        difference=difference,
    )
    db.add(close)
    db.commit()
    db.refresh(close)

    return render_template_string(
        CIERRE_CAJA_HTML,
        user=user,
        total_ventas=total_ventas,
        last_close=close,
        closed=True,
        difference=difference,
    )
